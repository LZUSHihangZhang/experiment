	\section{计算}
	氦氖激光器的波长为\[\lambda_{\text{标准}}=632.8 nm\]
	
	直接计算平均值
	\[\lambda = \frac{2\Delta d}{\Delta n} = \frac{2\times(\Delta d_1+\Delta d_2+\Delta d_3)}{30\times3} = 619.33mm\]
	相对误差 \[\frac{|619.33mm-632.8mm|}{632.8mm} \times 100\%=2.13\% \]
	逐差法
	\[\lambda = \frac{n_3+n_4-n_1-n_2}{(2\times30)^2} = 615.50mm\]
	相对误差 \[\frac{|615.5m-632.8mm|}{632.8mm} \times 100\%=2.73\% \]
	\section{误差分析}
	\[u_A = \sqrt{\frac{\sum_{i=1}^{N}(A_i-\bar{A})^2}{N(N-1)}}=\sqrt{\frac{\sum_{i=1}^{3}(\Delta d_i-\bar{\Delta d})^2}{3\times(3-1)}}=153.7nm\]
	\[u_B = \frac{1\times10^{-7}}{\sqrt{3}}m = 57.74nm\]
	\[u = \sqrt{u_A^2+u_B^2} = 163.605nm\]
	\[\frac{\partial \lambda}{\partial \Delta d_i} = \frac{1}{45}\]
	\[U = \sqrt{\sum_{i=1}^{3}\frac{\partial \lambda}{\partial \Delta d_i}\times u} = 3.3025769321445257nm\]
	则\[\lambda = (619.33\pm3.303)nm\]
